cmake_minimum_required(VERSION 3.31)
project(lux_tr01_harness VERSION 0.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "TR-01 native harness requires Windows x64")
endif()
add_executable(lux_native_smoke smoke.cc)
target_compile_features(lux_native_smoke PRIVATE cxx_std_20)
target_compile_options(lux_native_smoke PRIVATE /W4 /WX)
enable_testing()
add_test(NAME native_clock_smoke COMMAND lux_native_smoke)
add_executable(lux_pool_test ../tests/native/pool_test.cc)
target_include_directories(lux_pool_test PRIVATE texture-bridge/include)
target_compile_features(lux_pool_test PRIVATE cxx_std_20)
target_compile_options(lux_pool_test PRIVATE /UNDEBUG)
add_test(NAME pool_ownership COMMAND lux_pool_test)
add_library(lux_texture_bridge SHARED texture-bridge/src/electron_texture.cc)
target_include_directories(lux_texture_bridge PRIVATE vendor/node/node-v24.20.0/include/node)
target_compile_features(lux_texture_bridge PRIVATE cxx_std_20)
target_link_libraries(lux_texture_bridge PRIVATE d3d11 dxgi "${CMAKE_CURRENT_SOURCE_DIR}/vendor/node/node.lib")
set_target_properties(lux_texture_bridge PROPERTIES PREFIX "" SUFFIX ".node")
target_sources(lux_texture_bridge PRIVATE texture-bridge/src/delay_load.cc)
target_link_libraries(lux_texture_bridge PRIVATE delayimp)
target_link_options(lux_texture_bridge PRIVATE /DELAYLOAD:node.exe)
