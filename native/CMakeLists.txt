cmake_minimum_required(VERSION 3.31)
project(lux_tr01_harness VERSION 0.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "TR-01 native harness requires Windows x64")
endif()
add_executable(lux_native_smoke smoke.cc)
target_compile_features(lux_native_smoke PRIVATE cxx_std_20)
target_compile_options(lux_native_smoke PRIVATE /W4 /WX)
enable_testing()
add_test(NAME native_clock_smoke COMMAND lux_native_smoke)
